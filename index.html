This script uses the native browser `fetch` API to pull real-time data from GitHub and calculate the ship's hull strength (HP). ```javascript document.getElementById('fetchBtn').addEventListener('click', async () => { const username = document.getElementById('githubInput').value.trim(); const display = document.getElementById('statsDisplay'); if (!username) { display.innerHTML = " Please enter a username. "; return; } display.innerHTML = " Scanning GitHub databases... "; try { // Fetch data directly from GitHub's public API const response = await fetch(`https://api.github.com/users/${username}`); if (!response.ok) { throw new Error("Commander not found!"); } const data = await response.json(); // Calculate Ship Stats based on GitHub data const publicRepos = data.public_repos; const followers = data.followers; // Game Logic Formula: Base HP + (Repos * 10) + (Followers * 5) const hullStrength = 100 + (publicRepos * 10) + (followers * 5); // Display the results display.innerHTML = ` 📡 Link Established: @${data.login} Public Repositories: ${publicRepos} Followers: ${followers} Ship Hull Strength: ${hullStrength} HP `; } catch (error) { display.innerHTML = ` Error: ${error.message} `; } }); To test it, just double-click the `index.html` file so it opens in your web browser.
